📋 MediaWiki Setup Instructions

To set up MediaWiki for your FFXI server, follow these installation steps:

🐳 Docker Installation (Recommended)

# Clone MediaWiki Docker setup
git clone https://github.com/wikimedia/mediawiki-docker-dev.git wiki
cd wiki

# Create docker-compose.yml for FFXI wiki
cat > docker-compose.yml << 'EOF'
version: '3.8'
services:
  mediawiki:
    image: mediawiki:1.41
    container_name: ffxi-wiki
    ports:
      - "8080:80"
    environment:
      - MEDIAWIKI_DB_TYPE=mysql
      - MEDIAWIKI_DB_HOST=wiki-db
      - MEDIAWIKI_DB_NAME=ffxi_wiki
      - MEDIAWIKI_DB_USER=wiki_user
      - MEDIAWIKI_DB_PASSWORD=secure_password
      - MEDIAWIKI_SITE_NAME=FFXI Server Wiki
      - MEDIAWIKI_SITE_LANG=en
    volumes:
      - wiki_data:/var/www/html
      - ./LocalSettings.php:/var/www/html/LocalSettings.php
    depends_on:
      - wiki-db

  wiki-db:
    image: mariadb:10.11
    container_name: ffxi-wiki-db
    environment:
      - MYSQL_ROOT_PASSWORD=root_password
      - MYSQL_DATABASE=ffxi_wiki
      - MYSQL_USER=wiki_user
      - MYSQL_PASSWORD=secure_password
    volumes:
      - wiki_db:/var/lib/mysql

volumes:
  wiki_data:
  wiki_db:
EOF

# Start the wiki
docker-compose up -d

🔧 FFXI-Specific Configuration

# Add to LocalSettings.php for FFXI customization
$wgSitename = "FFXI Server Wiki";
$wgMetaNamespace = "FFXI";

# FFXI-themed skin
$wgDefaultSkin = "vector-2022";
$wgLogos = [
    '1x' => "$wgResourceBasePath/logo.png",
    'icon' => "$wgResourceBasePath/favicon.ico",
];

# Enable useful extensions
wfLoadExtension( 'Cite' );
wfLoadExtension( 'WikiEditor' );
wfLoadExtension( 'SyntaxHighlight_GeSHi' );
wfLoadExtension( 'TemplateData' );
wfLoadExtension( 'VisualEditor' );

# FFXI-specific namespaces
$wgExtraNamespaces[100] = "Quest";
$wgExtraNamespaces[101] = "Quest_talk";
$wgExtraNamespaces[102] = "Item";
$wgExtraNamespaces[103] = "Item_talk";
$wgExtraNamespaces[104] = "NPC";
$wgExtraNamespaces[105] = "NPC_talk";

# Database integration for live server data
$wgFFXIServerDB = [
    'host' => 'localhost',
    'database' => 'tpzdb',
    'username' => 'wiki_reader',
    'password' => 'secure_password'
];
